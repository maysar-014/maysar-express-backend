#### **Stock Reservation Engine:**
```typescript
// Advanced reservation with deadlock prevention and optimistic locking
export class ReservationEngine {
  async reserveStock(orderId: string, items: OrderItem[]): Promise<ReservationResult> {
    const reservations: Reservation[] = [];
    const rollbacks: (() => Promise<void>)[] = [];

    try {
      // Sort items by SKU for consistent locking order (deadlock prevention)
      const sortedItems = items.sort((a, b) => a.sku.localeCompare(b.sku));

      for (const item of sortedItems) {
        const warehouse = await this.selectOptimalWarehouse(item);
        const reservation = await this.createReservation(orderId, item, warehouse);

        reservations.push(reservation);
        rollbacks.push(() => this.rollbackReservation(reservation.id));
      }

      // All reservations successful - commit
      await this.commitReservations(reservations);
      return { success: true, reservations };

    } catch (error) {
      // Rollback all successful reservations
      await Promise.all(rollbacks.map(rollback => rollback()));
      throw new ReservationError('Stock reservation failed', { cause: error });
    }
  }

  private async selectOptimalWarehouse(item: OrderItem): Promise<Warehouse> {
    // Multi-factor warehouse selection:
    // 1. Stock availability
    // 2. Geographic proximity to shipping address
    // 3. Warehouse capacity and utilization
    // 4. Cost optimization
    // 5. Service level agreements

    const candidates = await this.findWarehousesWithStock(item.variantId, item.quantity);

    return candidates.reduce((best, current) => {
      const bestScore = this.calculateWarehouseScore(best, item);
      const currentScore = this.calculateWarehouseScore(current, item);
      return currentScore > bestScore ? current : best;
    });
  }
}
```

```typescript
// Continuous stock level monitoring with predictive analytics
export class StockMonitor {
  async monitorStockLevels(): Promise<void> {
    const lowStockItems = await this.identifyLowStockItems();

    for (const item of lowStockItems) {
      const urgency = this.calculateUrgency(item);
      const recommendation = await this.generateRecommendation(item);

      await this.emitLowStockAlert(item, urgency, recommendation);
      await this.triggerAutoReorder(item, recommendation);
    }
  }

  private calculateUrgency(item: StockItem): 'low' | 'medium' | 'high' | 'critical' {
    const stockRatio = item.quantityAvailable / item.reorderPoint;
    const salesVelocity = this.calculateSalesVelocity(item);
    const leadTime = this.getSupplierLeadTime(item);

    // Advanced urgency calculation considering:
    // - Current stock ratio
    // - Sales velocity trends
    // - Supplier lead times
    // - Seasonal demand patterns
    // - Alternative sourcing options

    if (stockRatio <= 0.1 || (salesVelocity > 10 && leadTime > 7)) return 'critical';
    if (stockRatio <= 0.25 || salesVelocity > 5) return 'high';
    if (stockRatio <= 0.5) return 'medium';
    return 'low';
  }
}
```

#### **Event-Driven Integration:**
```typescript
// Event handlers for seamless integration
export class InventoryEventHandlers {
  @EventHandler('OrderCreated.v1')
  async handleOrderCreated(event: OrderCreatedEvent): Promise<void> {
    // Reserve stock for new orders
    const reservation = await this.reservationEngine.reserveStock(
      event.payload.orderId,
      event.payload.items
    );

    // Emit reservation confirmation
    await this.eventBus.emit('StockReserved.v1', {
      orderId: event.payload.orderId,
      reservations: reservation.reservations
    });
  }

  @EventHandler('OrderCancelled.v1')
  async handleOrderCancelled(event: OrderCancelledEvent): Promise<void> {
    // Release reserved stock
    await this.reservationEngine.releaseStock(
      event.payload.orderId,
      'cancelled'
    );

    // Emit stock release event
    await this.eventBus.emit('StockReleased.v1', {
      orderId: event.payload.orderId,
      reason: 'cancelled'
    });
  }

  @EventHandler('ShipmentDelivered.v1')
  async handleShipmentDelivered(event: ShipmentDeliveredEvent): Promise<void> {
    // Update inventory levels
    await this.stockManager.reduceStock(
      event.payload.orderId,
      event.payload.items
    );

    // Emit inventory update events
    await this.eventBus.emit('InventoryUpdated.v1', {
      orderId: event.payload.orderId,
      items: event.payload.items
    });
  }
}
```

```typescript
// REST API with comprehensive validation and monitoring
export class InventoryApi {
  @Get('/stock/:variantId')
  @RateLimit({ windowMs: 60000, max: 100 })
  @Cache({ ttl: 300 })
  async getStockLevel(req: Request, res: Response): Promise<void> {
    const { variantId } = req.params;
    const { warehouseId } = req.query;

    const stock = await this.stockService.getStockLevel(variantId, warehouseId);

    res.json({
      variantId,
      warehouseId,
      available: stock.quantityAvailable,
      reserved: stock.quantityReserved,
      status: this.calculateStockStatus(stock),
      lastUpdated: stock.updatedAt
    });
  }

  @Post('/reservations')
  @ValidateBody(ReservationRequestSchema)
  @Transactional()
  async createReservation(req: Request, res: Response): Promise<void> {
    const { orderId, items } = req.body;

    const result = await this.reservationEngine.reserveStock(orderId, items);

    res.status(201).json({
      reservationId: result.reservationId,
      status: 'confirmed',
      expiresAt: result.expiresAt,
      items: result.items
    });
  }

  @Post('/inventory-counts')
  @RequireRole('inventory_manager')
  async startInventoryCount(req: Request, res: Response): Promise<void> {
    const { warehouseId, type, items } = req.body;

    const count = await this.inventoryCountService.startCount(
      warehouseId,
      type,
      req.user.id
    );

    res.status(201).json({
      countId: count.id,
      status: 'in_progress',
      estimatedCompletion: this.calculateEstimatedTime(items.length)
    });
  }
}
```

```sql
    overall_score DECIMAL(3,2) NOT NULL CHECK (overall_score >= 0 AND overall_score <= 5),
    review_count INTEGER DEFAULT 0,
    quality_score DECIMAL(3,2),
    delivery_score DECIMAL(3,2),
    communication_score DECIMAL(3,2),
    last_calculated TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    metadata JSONB,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE disputes (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    order_id UUID NOT NULL,
    buyer_id UUID NOT NULL,
    seller_id UUID NOT NULL,
    type VARCHAR(50) NOT NULL, -- item_not_received, item_damaged, wrong_item, buyer_remorse
    status VARCHAR(50) DEFAULT 'opened', -- opened, under_review, evidence_collected, decided, resolved
    priority VARCHAR(50) DEFAULT 'normal', -- low, normal, high, critical
    description TEXT NOT NULL,
    buyer_evidence JSONB,
    seller_evidence JSONB,
    admin_evidence JSONB,
    decision VARCHAR(50), -- buyer_wins, seller_wins, split, uphold
    refund_amount DECIMAL(10,2),
    resolution_notes TEXT,
    resolved_by UUID,
    resolved_at TIMESTAMP WITH TIME ZONE,
    sla_deadline TIMESTAMP WITH TIME ZONE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);
```
