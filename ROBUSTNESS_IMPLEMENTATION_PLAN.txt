DATABASE MIGRATION SCRIPTS:

```sql
-- Create comprehensive user table
CREATE TABLE users (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  email VARCHAR(255) UNIQUE NOT NULL,
  role user_role NOT NULL,
  status user_status DEFAULT 'PENDING',
  profile JSONB,
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

-- Create products table with full e-commerce features
CREATE TABLE products (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  seller_id UUID REFERENCES users(id),
  name VARCHAR(500) NOT NULL,
  description TEXT,
  price DECIMAL(10,2) NOT NULL,
  category_id UUID REFERENCES categories(id),
  inventory_count INTEGER DEFAULT 0,
  images JSONB,
  seo_data JSONB,
  status product_status DEFAULT 'DRAFT',
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

-- Create orders table with complete lifecycle
CREATE TABLE orders (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  buyer_id UUID REFERENCES users(id),
  seller_id UUID REFERENCES users(id),
  status order_status DEFAULT 'PENDING',
  total_amount DECIMAL(10,2) NOT NULL,
  items JSONB NOT NULL,
  shipping_address JSONB,
  payment_status payment_status DEFAULT 'PENDING',
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);
```

/services/compliance-service/src/localCompliance.ts
export class LocalComplianceService {
  async calculateComplianceScore(authority: string): Promise<number> {
    const requirements = await this.getRequirements(authority);
    const completedRequirements = requirements.filter(r => r.status === 'met');
    const partialRequirements = requirements.filter(r => r.status === 'partial');
    
    // Calculate weighted score
    const baseScore = (completedRequirements.length / requirements.length) * 100;
    const partialPenalty = (partialRequirements.length / requirements.length) * 5;
    
    return Math.max(0, Math.min(100, baseScore - partialPenalty));
  }

  async getComplianceStatus(): Promise<ComplianceMetric[]> {
    const authorities = ['TCRA', 'TRA', 'BOT', 'FIU', 'BRELA', 'TCAA'];
    const metrics = [];

    for (const authority of authorities) {
      const score = await this.calculateComplianceScore(authority);
      const status = this.determineStatus(score);
      const requirements = await this.getRequirements(authority);
      
      metrics.push({
        authority,
        fullName: this.getAuthorityFullName(authority),
        complianceScore: Math.round(score * 10) / 10,
        status,
        requirements,
        lastCalculated: new Date()
      });
    }

    return metrics;
  }

  private determineStatus(score: number): 'compliant' | 'partial' | 'non_compliant' | 'critical' {
    if (score >= 95) return 'compliant';
    if (score >= 85) return 'partial';
    if (score >= 70) return 'non_compliant';
    return 'critical';
  }

  private async getRequirements(authority: string): Promise<ComplianceRequirement[]> {
    // Dynamic requirement checking based on actual system state
    switch (authority) {
      case 'TCRA':
        return [
          { id: 'TCRA-001', description: 'Service provider registration', status: await this.checkServiceRegistration() },
          { id: 'TCRA-002', description: 'Data protection compliance', status: await this.checkDataProtection() },
          { id: 'TCRA-003', description: 'Consumer protection', status: await this.checkConsumerProtection() }
        ];
      case 'TRA':
        return [
          { id: 'TRA-001', description: 'VAT registration', status: await this.checkVATRegistration() },
          { id: 'TRA-002', description: 'Electronic fiscal device', status: await this.checkFiscalDevice() },
          { id: 'TRA-003', description: 'Digital service tax', status: await this.checkDigitalServiceTax() }
        ];
      case 'BOT':
        return [
          { id: 'BOT-001', description: 'Payment service provider license', status: await this.checkPSPLicense() },
          { id: 'BOT-002', description: 'AML compliance', status: await this.checkAMLCompliance() },
          { id: 'BOT-003', description: 'KYC requirements', status: await this.checkKYCRequirements() }
        ];
      default:
        return [];
    }
  }

  // Dynamic requirement checkers
  private async checkServiceRegistration(): Promise<'met' | 'partial' | 'not_met'> {
    const registration = await prisma.serviceRegistration.findFirst({ where: { authority: 'TCRA' } });
    if (!registration) return 'not_met';
    if (registration.expiryDate < new Date()) return 'partial';
    return 'met';
  }

  private async checkDataProtection(): Promise<'met' | 'partial' | 'not_met'> {
    const gdprCompliance = await this.checkGDPRImplementation();
    const dataRetentionPolicies = await this.checkDataRetentionPolicies();
    
    if (gdprCompliance && dataRetentionPolicies) return 'met';
    if (gdprCompliance || dataRetentionPolicies) return 'partial';
    return 'not_met';
  }

  private async checkVATRegistration(): Promise<'met' | 'partial' | 'not_met'> {
    const vatCertificate = await prisma.taxCertificate.findFirst({ 
      where: { type: 'VAT', status: 'active' } 
    });
    return vatCertificate ? 'met' : 'not_met';
  }

  private async checkPSPLicense(): Promise<'met' | 'partial' | 'not_met'> {
    const license = await prisma.paymentLicense.findFirst({ 
      where: { authority: 'BOT', status: 'active' } 
    });
    if (!license) return 'not_met';
    if (license.status === 'pending') return 'partial';
    return 'met';
  }
}

export const authenticateUser = async (req: Request, res: Response, next: NextFunction) => {
  try {
    const token = extractTokenFromHeader(req.headers.authorization);
    if (!token) {
      return res.status(401).json({ error: 'Authentication required' });
    }

    const payload = await verifyJWTToken(token);
    const user = await getUserById(payload.sub);
    
    if (!user || user.status !== 'ACTIVE') {
      return res.status(401).json({ error: 'Invalid user' });
    }

    req.user = user;
    req.permissions = ROLE_PERMISSIONS[user.role];
    next();
  } catch (error) {
    return res.status(401).json({ error: 'Invalid token' });
  }
};

export const requirePermission = (permission: Permission) => {
  return (req: Request, res: Response, next: NextFunction) => {
    if (!req.permissions?.includes(permission)) {
      return res.status(403).json({ error: 'Insufficient permissions' });
    }
    next();
  };
};

export const requireRole = (roles: UserRole[]) => {
  return (req: Request, res: Response, next: NextFunction) => {
    if (!roles.includes(req.user.role)) {
      return res.status(403).json({ error: 'Insufficient role' });
    }
    next();
  };
};

  const [showMFA, setShowMFA] = useState(false);

  const handleLogin = async (e: FormEvent) => {
    e.preventDefault();
    
    try {
      const { data, error } = await supabase.auth.signInWithPassword({
        email,
        password
      });

      if (error) throw error;

      // Check if MFA is required
      if (data.user?.user_metadata?.mfa_enabled) {
        setShowMFA(true);
        return;
      }

      // Redirect based on user role
      const userRole = data.user?.user_metadata?.role;
      redirectToConsole(userRole, redirectTo);
    } catch (error) {
      setError(error.message);
    }
  };

  return (
    <form onSubmit={handleLogin} className="auth-form">
      <input
        type="email"
        value={email}
        onChange={(e) => setEmail(e.target.value)}
        placeholder="Email"
        required
      />
      <input
        type="password"
        value={password}
        onChange={(e) => setPassword(e.target.value)}
        placeholder="Password"
        required
      />
      
      {showMFA && (
        <input
          type="text"
          value={mfaCode}
          onChange={(e) => setMfaCode(e.target.value)}
          placeholder="MFA Code"
          required
        />
      )}
      
      <button type="submit">Login</button>
    </form>
  );
};

// Permission-Based Component Wrapper
/shared/components/auth/PermissionGuard.tsx
interface PermissionGuardProps {
  permission?: Permission;
  role?: UserRole[];
  fallback?: React.ReactNode;
  children: React.ReactNode;
}

export const PermissionGuard: React.FC<PermissionGuardProps> = ({
  permission,
  role,
  fallback = null,
  children
}) => {
  const { user, permissions } = useAuth();

  const hasPermission = () => {
    if (permission && !permissions.includes(permission)) {
      return false;
    }
    if (role && !role.includes(user.role)) {
      return false;
    }
    return true;
  };

  if (!hasPermission()) {
    return <>{fallback}</>;
  }

  return <>{children}</>;
};

// Console-Specific Auth Hook
/shared/hooks/useConsoleAuth.ts
export const useConsoleAuth = (requiredRole: UserRole) => {
  const { user, loading, error } = useAuth();
  const router = useRouter();

  useEffect(() => {
    if (!loading && (!user || user.role !== requiredRole)) {
      router.push('/auth/login?redirect=' + encodeURIComponent(router.asPath));
    }
  }, [user, loading, requiredRole, router]);

  const hasPermission = (permission: Permission): boolean => {
    return user?.permissions?.includes(permission) || false;
  };

  const canAccess = (resource: string, action: string): boolean => {
    const permission = `${resource}:${action}` as Permission;
    return hasPermission(permission);
  };

  return {
    user,
    loading,
    error,
    hasPermission,
    canAccess,
    isAuthenticated: !!user && user.role === requiredRole
  };
};
